pymongo==4.19.0
# Required to resolve a `mongodb+srv://` URI, which is the only form DigitalOcean
# hands out for a managed cluster. pymongo does not depend on it, so without this
# the image builds and then fails at connect time with "dnspython must be
# installed" — a fault that cannot appear until deploy.
dnspython==2.9.0
pydantic==2.14.1
pydantic-settings==2.16.0
email-validator==2.3.0
PyJWT==2.15.1
# `zoneinfo` reads the operating system's tz database, which Windows does not
# have and the slim base image does not install. Emails render dates and times
# in DISPLAY_TIMEZONE, so without this every receipt silently falls back to UTC
# — five and a half hours out for an Indian customer, and wrong on the day.
tzdata==2026.3
